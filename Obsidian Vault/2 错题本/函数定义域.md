```tikz
\begin{document}

\definecolor{qqwuqq}{rgb}{0,0.39,0}
\begin{tikzpicture}[x=1cm,y=1cm,scale=2]
    % 坐标轴：y轴拉长到2.2，装下f(0)=2
    \draw[->] (-1.2,0) -- (1.2,0) node[right,font=\small] {$x$};
    \draw[->] (0,-1.2) -- (0,1.2) node[above,font=\small] {$lnx$};
    
    % x≠0时的sin(1/x)曲线（采样数100、终止点0.02，符合可渲染规则）
    \draw[line width=0.6pt,color=qqwuqq,smooth,samples=100,domain=-1:-0.02] 
        plot(\x,{sin((1/(\x))*180/pi)});
    \draw[line width=0.6pt,color=qqwuqq,smooth,samples=100,domain=0.02:1] 
        plot(\x,{ln(\x)});
    
    % x=0时的实心点（f(0)=2）
    \fill[qqwuqq] (0,1.1) circle (1.2pt);
    
    % 分段函数标注：用纯基础LaTeX语法（不用amsmath的cases）
    \draw[color=qqwuqq,font=\small] (1.0, 1.8) node{$\displaystyle
    f(x)=\left\{ \begin{array}{ll}
    \sin\frac{1}{x}, & x\neq0 \\[4pt]
    2, & x=0
    \end{array} \right.$};
\end{tikzpicture}

\end{document}
```
