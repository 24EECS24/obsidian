`````col
==分析：==
$\displaystyle  由\frac{1}{\sqrt{4-x^2}}得4-x^2>0$
```tikz
\begin{document}

\definecolor{qqwuqq}{rgb}{0,0.39,0}
\begin{tikzpicture}[x=1cm,y=1cm,scale=0.4]
    % 坐标轴：y轴拉长到2.2，装下f(0)=2
    \draw[->] (-6,0) -- (6,0) node[right,font=\large] {$x$};
    \draw[->] (0,-6) -- (0,6) node[above,font=\large] {$y=-x^2+4$};
    
    % x≠0时的sin(1/x)曲线（采样数100、终止点0.02，符合可渲染规则）
    \draw[line width=0.6pt,color=qqwuqq,smooth,samples=100,domain=-3:0] 
        plot(\x,{-(\x)^2+4});
    \draw[line width=0.6pt,color=qqwuqq,smooth,samples=100,domain=0:3] 
        plot(\x,{-(\x)^2+4});
        
    \fill[qqwuqq] (0,4) circle (4pt);
    \draw[qqwuqq] (-2,0) circle (4pt);
    \draw[qqwuqq] (2,0) circle (4pt);
    \draw[color=qqwuqq,font=\large] (2.5,0.5) node{$2$};
    \draw[color=qqwuqq,font=\large] (-4,0.5) node[right,font=\large] {$-2$};
    \draw[color=qqwuqq,font=\large] (0,4.3) node[right,font=\large] {$4$};
    
\end{tikzpicture}

\end{document}
`````
