\begin{align*}  % *号表示不自动加公式编号，更简洁
&设函数列\left \{ u_n(x) \right \} 定义在区间I上，称\\[4pt]
&u_1(x)+u_2(x)+u_3(x)+\dots +u_n(x)+\dots   \\[4pt]
&为定义在区间I上的函数项级数，\\[4pt]
&记为\sum_{n=0}^{\infty} u_n(x)\\[4pt]
&当x取确定的值x_0时，  \\[4pt]
&\sum_{n=0}^{\infty} u_n(x)成为常数项级数\sum_{n=0}^{\infty} u_n(x_0)\\[4pt]
\end{align*}
$$
